"""TARGET-LOOK demo: the skid shape scaled (SKID_V valve relays, SKID_M fan starters).

Same shapes as skid.py, more of them, so groups spill over pages. Demo parts only.
"""

import math
import os
import tempfile
from pathlib import Path

import fransys as fr
from fransys.colours import BK, BU

V = int(os.environ.get("SKID_V", "10"))
M = int(os.environ.get("SKID_M", "3"))


def design():
    d = fr.design("demo_parts", place="CAB")
    d.project(title="Skid", number="P-1", customer="Example Co", revision=1, author="OJB")
    d.revision(1, date="2026-10-15", text="First issue", created="XX")
    cab = d.location("CAB", "Cabinet")
    w = {"wire": (BU, 0.5)}
    n24 = 4 + 2 * M + 2 * V
    with d.function("G0", "Supply 24 V"):
        live = d.terminal_strip("X10", "DEMO-TB-2.5", n24).run("P", n24, bridged=True)
        rail = d.terminal_strip("X11", "DEMO-TB-2.5", n24).run("M", n24, bridged=True)
        d.dc_supply("S", plus=live[1], minus=rail[1], voltage=24)
    li = iter(range(2, n24 + 1))
    ri = iter(range(2, n24 + 1))
    with d.function("G1", "Mains"):
        q0 = d.device("Q0", "DEMO-MCB-3P")
        x3 = d.terminal_strip("X3", "DEMO-TB-2.5", 3 * M)
    starters = []
    with d.function("G2", "Fan starters"):
        for s in range(M):
            f = d.device(f"F{3 + s}", "DEMO-MCB-3P")
            q = d.device(f"Q{3 + s}", "DEMO-CTR-3P-NC")
            ol = d.device(f"F{13 + s}", "DEMO-OVERLOAD-3P")
            mo = d.device(f"M{1 + s}", "DEMO-MOTOR-4KW")
            h = d.device(f"H{3 + s}", "DEMO-LAMP-24")
            for i, (a, b) in enumerate((("1", "2"), ("3", "4"), ("5", "6"))):
                t = x3[3 * s + i + 1]
                src = q0.element[b] if s == 0 else x3[3 * (s - 1) + i + 1].inner
                d.detail.wire(src, t.inner, wire=(BK, 1.5))
                d.detail.wire(t.outer, f.element[a], wire=(BK, 1.5))
                d.detail.wire(f.element[b], q.main[a], wire=(BK, 1.5))
                d.detail.wire(q.main[b], ol.main[a], wire=(BK, 1.5))
                d.detail.wire(ol.main[b], mo.motor["UVW"[i]], wire=(BK, 1.5))
            d.detail.wire(live[next(li)].outer, q.aux["13"], **w)
            d.detail.wire(q.aux["14"], h.lamp["1"], **w)
            d.detail.wire(h.lamp["2"], rail[next(ri)].outer, **w)
            d.detail.wire(q.coil["A2"], rail[next(ri)].outer, **w)
            starters.append((q, ol))
    nch = M + V
    with d.function("G3", "PLC"):
        di = d.device("DI1", "DEMO-PLC-DI-16")
        d.device("AI1", "DEMO-PLC-AI-4")
        d.device("AI2", "DEMO-PLC-AI-4")
        dos = [d.device(f"DO{i + 1}", "DEMO-PLC-DO-2") for i in range(math.ceil(nch / 2))]
    chans = [getattr(dos[c // 2], f"do_{c % 2 + 1}")[str(c % 2 + 1)] for c in range(nch)]
    with d.function("G4", "Valve relays"):
        f8 = d.device("F8", "DEMO-MCB-C6")
        d.detail.wire(live[next(li)].outer, f8.element["1"], **w)
        relays = [d.device(f"K{i + 4}", "DEMO-RLY-2CO-24") for i in range(V)]
    with d.function("G5", "Valves"):
        valves = [d.device(f"Y{i + 4}", "DEMO-LAMP-24") for i in range(V)]
    for i, (k, y) in enumerate(zip(relays, valves, strict=True)):
        d.detail.wire(chans[M + i], k.coil["A1"], **w)
        d.detail.wire(k.coil["A2"], rail[next(ri)].outer, **w)
        d.detail.wire(f8.element["2"], k.co_1["11"], **w)
        d.detail.wire(k.co_1["14"], y.lamp["1"], **w)
        d.detail.wire(y.lamp["2"], rail[next(ri)].outer, **w)
    for s, (q, ol) in enumerate(starters):
        d.detail.wire(chans[s], q.coil["A1"], **w)
    fb = []
    for q, ol in starters:
        fb += [(q.aux_nc, "21", "22"), (ol.aux, "95", "96")]
    fb += [(k.co_2, "21", "24") for k in relays]
    for i, (fun, a, b) in enumerate(fb[:16]):
        d.detail.wire(live[next(li)].outer, fun[a], **w)
        d.detail.wire(fun[b], getattr(di, f"di_{i + 1}")[str(i + 1)], **w)
    cover = Path(tempfile.mkdtemp()) / "cover.md"
    cover.write_text("# Skid\n", encoding="utf-8")
    return d, fr.document(fr.DocumentPreset.CABINET_SCHEMATIC, cab, cover=cover)
